import datetime
import os
import requests

SAVE_DIR_DEFAULT = (
    r"K:\Sagar_new\ORDER\00_Programming\python\projects"
    r"\stocktracker_revamp\files\bhavcopy"
)

# ----------------------------------------------------------------------
# Build URL for Full-Bhavcopy + Security-Deliverable file
# ----------------------------------------------------------------------
def _make_full_url(date: datetime.date) -> str:
    """Return CDN URL for sec_bhavdata_full_<DDMMYYYY>.csv"""
    month_abbr = date.strftime("%b").upper()             # JUN, JUL, etc.
    return (
        "https://archives.nseindia.com/content/historical/EQUITIES/"
        f"{date:%Y}/{month_abbr}/sec_bhavdata_full_{date:%d%m%Y}.csv"
    )

# ----------------------------------------------------------------------
# Download routine (same browser-style Session you’re already using)
# ----------------------------------------------------------------------
def download_full_bhavcopy(
        date: datetime.date,
        save_dir: str = SAVE_DIR_DEFAULT
) -> str:
    """
    Fetch the *Full Bhavcopy + Security Deliverable* CSV for `date`
    and save it under `save_dir`.  Returns the local file path.
    """
    url = _make_full_url(date)

    session = requests.Session()
    session.headers.update({
        "User-Agent": (
            "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
            "AppleWebKit/537.36 (KHTML, like Gecko) "
            "Chrome/124.0.0.0 Safari/537.36"
        ),
        "Accept": "text/csv,*/*;q=0.9",
        "Accept-Language": "en-US,en;q=0.9",
        "Accept-Encoding": "gzip, deflate, br",
        "Referer": "https://www.nseindia.com/",
        "DNT": "1",
        "Connection": "keep-alive",
    })

    resp = session.get(url, timeout=30)
    resp.raise_for_status()           # clear failure if file missing

    os.makedirs(save_dir, exist_ok=True)
    local_path = os.path.join(save_dir, url.rsplit("/", 1)[-1])

    with open(local_path, "wb") as f:
        f.write(resp.content)

    print(f"✅  Saved → {local_path}")
    return local_path


# --------------------------- quick demo ----------------------------
if __name__ == "__main__":
    # grabs 4 Jun 2025; change the date as required
    download_full_bhavcopy(datetime.date(2025, 6, 4))
